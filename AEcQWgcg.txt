    localStorage.setItem(
      EFFORT_KEY,
      JSON.stringify(all)
    );

  function getBackendModelId(uiModelId) {
    if (!uiModelId) return null;

    const selected = modelById(uiModelId);

    if (
      selected &&

    }

    return uiModelId;

  // ---------------------------------------------------------------------------

  const originalFetch = window.fetch;

  window.fetch = async function (input, init) {
    const url =
      typeof input === 'string'
        ? input
        : (input && input.url) || '';

    let patchedConfig = false;

    try {
      const selectedUiModel =
        selectedModelId();

      const model =
        getBackendModelId(selectedUiModel);

      const effort =
        currentEffort();

      if (
        model &&
        init &&
        typeof init.body === 'string' &&
        init.body.includes('"config"')
      ) {
        const body =
          JSON.parse(init.body);

        let changed = false;

        const patchConfig = (value) => {
          if (
            !value ||
            typeof value !== 'object'
          ) {
            return;
          }

          if (
            value.type === 'config' &&
            value.value &&
            typeof value.value === 'object' &&
            'model' in value.value
          ) {
            value.value.model = model;
            value.value.modelFromUser = true;

            if (effort) {
              value.value.reasoningEffort =
                effort;

              value.value.effort =
                effort;
            }

            changed = true;
          }

          for (
            const child of
            Object.values(value)
          ) {
            if (Array.isArray(child)) {
              child.forEach(patchConfig);
            } else if (
              child &&
              typeof child === 'object'
            ) {
              patchConfig(child);
            }
          }
        };

        patchConfig(body);

        if (changed) {
          init = {
            ...init,
            body: JSON.stringify(body),
          };

          patchedConfig = true;

          console.log(
            '[ModelPicker] [Request Patched] URL:',
            url,
            '| model:',
            model,
            '| effort:',
            effort || 'default',
            '| Body:',
            body
          );
        }
      }
    } catch (error) {
      console.warn(
        '[ModelPicker] request patch failed:',
        error
      );
    }

    const response =
      await originalFetch.call(
        this,
        input,
        init
      );

    const isModelUrl =
      /availablemodels/i.test(url) ||
      /getavailablemodels/i.test(url) ||
      /getaiavailablemodels/i.test(url);

    if (isModelUrl) {
      const statusInfo = `[Status: ${response.status} ${response.statusText}]`;
      console.log(`[ModelPicker] [Fetch Response - Models] ${url} ${statusInfo}`);
      try {
        const clone = response.clone();
        const contentType = clone.headers.get('content-type') || '';

        if (contentType.includes('application/json')) {
          clone.json().then((json) => {
            console.log(
              `[ModelPicker] [Fetch Response Data (JSON)] ${url}:`,
              json
            );
            harvestModels(json);
          }).catch((err) => {
            console.warn(`[ModelPicker] Failed to parse models JSON from ${url}:`, err);
          });
        } else {
          clone.text().then((text) => {
            let parsed = null;
            try {
              parsed = JSON.parse(text);
            } catch {}
            console.log(
              `[ModelPicker] [Fetch Response Data] ${url}:`,
              parsed || text
            );
            if (parsed) {
              harvestModels(parsed);
            }
          }).catch((err) => {
            console.warn(`[ModelPicker] Failed to read models text from ${url}:`, err);
          });
        }
      } catch (error) {
        console.warn(
          `[ModelPicker] Could not clone/read response from ${url}:`,
          error
        );
      }
    } else if (
      patchedConfig ||
      /\/api\/v3\/(runAi|getAi|ai)/i.test(url)
    ) {
      const statusInfo = `[Status: ${response.status} ${response.statusText}]`;
      console.log(
        `[ModelPicker] [Fetch Response - AI Endpoint] ${url} ${statusInfo}`
      );
      try {
        const clone = response.clone();
        const contentType = clone.headers.get('content-type') || '';

        if (contentType.includes('application/json')) {
          clone.json().then((json) => {
            console.log(
              `[ModelPicker] [Fetch Response AI Data (JSON)] ${url}:`,
              json
            );
          }).catch((e) => {
            console.warn(
              `[ModelPicker] Could not parse AI response JSON from ${url}:`,
              e
            );
          });
        } else {
          clone.text().then((text) => {
            let parsed = null;
            try {
              parsed = JSON.parse(text);
            } catch {}
            console.log(
              `[ModelPicker] [Fetch Response AI Data] ${url} (length: ${text.length}):`,
              parsed || (text.length > 5000 ? text.slice(0, 5000) + '... [truncated]' : text)
            );
          }).catch((e) => {
            console.warn(
              `[ModelPicker] Could not read AI response text from ${url}:`,
              e
            );
          });
        }
      } catch (err) {
        console.warn(
          `[ModelPicker] Could not clone AI response from ${url}:`,
          err
        );
      }
    }

    return response;
  };

  // ---------------------------------------------------------------------------
  // XHR interception
  // ---------------------------------------------------------------------------

  const XhrOpen =
    XMLHttpRequest.prototype.open;

  XMLHttpRequest.prototype.open =
    function (
      method,
      requestUrl,
      ...rest
    ) {
      this.__naiUrl =
        String(requestUrl || '');
      this.__naiMethod =
        String(method || 'GET');

      return XhrOpen.call(
        this,
        method,
        requestUrl,
        ...rest
      );
    };

  const XhrSend =
    XMLHttpRequest.prototype.send;

  XMLHttpRequest.prototype.send =
    function (...args) {
      const url = String(this.__naiUrl || '');
      const isModelRequest =
        /availablemodels/i.test(url) ||
        /getavailablemodels/i.test(url) ||
        /getaiavailablemodels/i.test(url);

      if (isModelRequest) {
        console.log(
          `[ModelPicker] [XHR Request] ${this.__naiMethod} ${url} | Payload:`,
          args[0]
        );
      }

      this.addEventListener(
        'load',
        () => {
          if (!isModelRequest) {
            return;
          }

          console.log(
            `[ModelPicker] [XHR Response] ${url} [Status: ${this.status} ${this.statusText}]`
          );

          try {
            const data = JSON.parse(this.responseText);
            console.log(
              `[ModelPicker] [XHR Response Data (JSON)] ${url}:`,
              data
            );
            harvestModels(data);
          } catch (err) {
            console.log(
              `[ModelPicker] [XHR Response Raw Text] ${url}:`,
              this.responseText
            );
          }
        }
      );

      this.addEventListener(
        'error',
        (err) => {
          if (isModelRequest) {
            console.error(
              `[ModelPicker] [XHR Network Error] ${url}:`,
              err
            );
          }
        }
      );

      return XhrSend.apply(
        this,
        args
      );
    };

  // ---------------------------------------------------------------------------
  // Probe model list
  // ---------------------------------------------------------------------------

  async function probeModelList() {
    if (MODELS && MODELS.length) {
      console.log(
        '[ModelPicker] [Probe] Models are embedded directly in code (' +
          MODELS.length +
          ' models). Network probing skipped.'
      );
      return;
    }

    if (probing) return;

    probing = true;
    console.log('[ModelPicker] [Probe] Starting probeModelList to fetch available models from server...');

    const paths = [
      '/api/v3/getAvailableModels',
      '/api/v3/getAiAvailableModels',
      '/api/v3/ai/getAvailableModels',
    ];

    const guessedSpaceId = spaceIdGuess();
    console.log('[ModelPicker] [Probe] Guessed spaceId:', guessedSpaceId);

    for (const path of paths) {
      for (
        const body of [
          '{}',
          JSON.stringify({
            spaceId: guessedSpaceId,
          }),
        ]
      ) {
        try {
          console.log(`[ModelPicker] [Probe Request] POST ${path} | Body:`, body);
          const response =
            await originalFetch.call(
              window,
              path,
              {
                method: 'POST',
                credentials: 'include',
                headers: {
                  'content-type':
                    'application/json',
                },
                body,
              }
            );

          let respText = '';
          try {
            respText = await response.text();
          } catch (textErr) {
            console.warn(`[ModelPicker] [Probe] Could not read response text from ${path}:`, textErr);
          }

          let data = null;
          try {
            data = JSON.parse(respText);
          } catch {
            data = null;
          }

          console.log(
            `[ModelPicker] [Probe Response] POST ${path} [Status: ${response.status} ${response.statusText}]:`,
            data !== null ? data : respText
          );

          if (!response.ok) {
            console.warn(
              `[ModelPicker] [Probe] Server returned non-OK status ${response.status} for ${path}:`,
              data !== null ? data : respText
            );
            continue;
          }

          if (data) {
            const before =
              MODELS.length;

            harvestModels(data);

            if (
              MODELS.length > before ||
              MODELS.length
            ) {
              console.log(
                '[ModelPicker] [Probe Success] Model list successfully loaded from',
                path,
                '| Total models:',
                MODELS.length
              );

              probing = false;
              return;
            } else {
              console.warn(
                `[ModelPicker] [Probe] Server returned OK from ${path}, but no models could be parsed.`
              );
            }
          }
        } catch (err) {
          console.error(`[ModelPicker] [Probe Error] Network exception for ${path}:`, err);
        }
      }
    }

    console.warn(
      '[ModelPicker] Could not load the model list from probe paths; reload the AI chat'
    );

    probing = false;
  }

  function spaceIdGuess() {
    const match =
      document.cookie.match(
        /spaceId=([0-9a-f-]{36})/i
      );

    if (match) return match[1];

    for (
      let index = 0;
      index < localStorage.length;
      index += 1
    ) {
      const key =
        localStorage.key(index) || '';

      const found =
        key.match(
          /([0-9a-f]{8}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{4}-[0-9a-f]{12})/i
        );

      if (found) {
        return found[1];
      }
    }

    return undefined;
  }

  function findModelArray(
    value,
    seen = new WeakSet()
  ) {
    if (
      !value ||
      typeof value !== 'object' ||
      seen.has(value)
    ) {
      return null;
    }

    seen.add(value);

    if (
      Array.isArray(value.models) &&
      value.models.some(
        (item) =>
          item &&
          typeof item === 'object' &&
          item.model
      )
    ) {
      return value.models;
    }

    for (
      const child of Object.values(value)
    ) {
      if (
        child &&
        typeof child === 'object'
      ) {
        const found =
          findModelArray(
            child,
            seen
          );

        if (found) {
          return found;
        }
      }
    }

    return null;
  }

  function providerLabel(item) {
    const raw =
      String(
        item.modelProvider ||
        item.modelFamily ||
        item.provider ||
        'Other'
      ).trim();

    const known = {
      anthropic: 'Claude',
      claude: 'Claude',
      openai: 'GPT',
      gpt: 'GPT',
      google: 'Gemini',
      gemini: 'Gemini',
      moonshot: 'Kimi',
      kimi: 'Kimi',
      fireworks: 'Other',
    };

    return (
      known[
        raw.toLowerCase()
      ] || raw
    );
  }

  function harvestModels(data) {
    console.log(
      '[ModelPicker] [harvestModels] Server response data received:',
      data
    );

    const list =
      findModelArray(data);

    if (
      !list ||
      !list.length
    ) {
      console.warn(
        '[ModelPicker] [harvestModels] No models found in server response data:',
        data
      );
      return;
    }

    console.log(
      '[ModelPicker] [harvestModels] Extracted models array (' +
        list.length +
        ' items):',
      list
    );

    const unique =
      new Map();

    for (
      const item of list
    ) {
      if (
        !item ||
        !item.model
      ) {
        continue;
      }

      const config =
        item.modelConfiguration ||
        {};

      const modelId =
        String(item.model);

      const modelLabel =
        String(
          item.modelMessage ||
          item.displayName ||
          item.model
        );

      const isAstra =
        /astra/i.test(modelId) ||
        /astra/i.test(modelLabel);

      unique.set(
        modelId,
        {
          id: modelId,

          label: modelLabel,

          provider:
            providerLabel(item),

          /*
           * All models are enabled and selectable in the picker.
           */
          disabled: false,

          reason: '',

          uiSelectable: true,

          efforts:
            Array.isArray(
              config.supportedReasoningEfforts
            )
              ? config
                  .supportedReasoningEfforts
                  .filter(Boolean)
              : [],

          defaultEffort:
            config.defaultReasoningEffort ||
            null,

          stats:
            item.modelCardAttributes ||
            null,
        }
      );
    }

    MODELS =
      [...unique.values()].sort(
        (a, b) =>
          Number(a.disabled) -
            Number(b.disabled) ||
          a.provider.localeCompare(
            b.provider
          ) ||
          a.label.localeCompare(
            b.label
          )
      );

    try {
      localStorage.setItem(
        CACHE_KEY,
        JSON.stringify(MODELS)
      );
    } catch {
      /* ignore */
    }

    console.log(
      '[ModelPicker] [harvestModels] Successfully loaded',
      MODELS.length,
      'models (' +
        MODELS.filter((m) => !m.disabled).length +
        ' available, ' +
        MODELS.filter((m) => m.disabled).length +
        ' disabled):',
      MODELS
    );

    renderModelList();
    renderEffortMenu();
    updateControls();
  }

  function statsSummary(model) {
    if (!model.stats) return '';

    const parts = [];

    const {
      intelligence,
      speed,
      cost,
    } = model.stats;

    if (
      intelligence != null
    ) {
      parts.push(
        `smarts ${intelligence}/5`
      );
    }

    if (speed != null) {
      parts.push(
        `speed ${speed}/5`
      );
    }

    if (cost != null) {
      parts.push(
        `cost ${cost}/5`
      );
    }

    return parts.join(' · ');
  }

  function renderEffortMenu() {
    const list =
      document.getElementById(
        'nai-effort-list'
      );

    if (!list) return;

    list.replaceChildren();

    if (!selectedModelId()) {
      const note =
        document.createElement(
          'div'
        );

      note.className =
        'nai-empty';

      note.textContent =
        'Pick a model first. Auto mode is controlled by Notion.';

      list.appendChild(
        note
      );

      return;
    }

    const options =
      availableEfforts();

    const active =
      currentEffort();

    const fallback =
      defaultEffort();

    for (
      const effort of options
    ) {
      list.appendChild(
        buildRow({
          name:
            titleCase(effort),

          badge:
            effort === fallback
              ? 'Default'
              : '',

          active:
            effort === active,

          onSelect: () => {
            setEffort(effort);

            console.log(
              '[ModelPicker] effort:',
              effort
            );

            renderEffortMenu();
            updateControls();
            closeAll();
          },
        })
      );
    }
  }

          },
        })
      );

      const separator =
        document.createElement(
          'div'
        );

      separator.className =
        'nai-separator';

      list.appendChild(
        separator
      );
    }

    const visible =
      MODELS.filter(
        (model) =>
          `${model.label} ${model.provider} ${model.id}`
            .toLocaleLowerCase()
            .includes(query)
      );

    const groups =
      new Map();

    for (
      const model of visible
    ) {
      if (
        !groups.has(
          model.provider
        )
      ) {
        groups.set(
          model.provider,
          []
        );
      }

      groups
        .get(model.provider)
        .push(model);
    }

    for (
      const [
        provider,
        models,
      ] of groups
    ) {
      const heading =
        document.createElement(
          'div'
        );

      heading.className =
        'nai-section-label';

      heading.textContent =
        provider;

      list.appendChild(
        heading
      );

      for (
        const model of models
      ) {
        const metaBits = [];

        if (
          model.efforts.length
        ) {
          metaBits.push(
            `effort: ${model.efforts.join(', ')}`
          );
        }

        const stats =
          statsSummary(model);

        if (stats) {
          metaBits.push(stats);
        }

        /*
         * All models are UI-selectable.
         */
        const uiDisabled = false;

        list.appendChild(
          buildRow({
            icon:
              iconFor(model),

            name:
              model.label,

            meta:
              uiDisabled
                ? (
                    model.reason ||
                    'Unavailable on this plan'
                  )
                : (
                    metaBits.join(
                      ' · '
                    ) ||
                    model.id
                  ),

            active:
              selectedModelId() ===
              model.id,

            disabled:
              uiDisabled,

            title:
              uiDisabled
                ? model.reason
                : '',

            onSelect: () => {
              localStorage.setItem(
                KEY,
                model.id
              );

              console.log(
                '[ModelPicker] selected:',
                model.label
              );

              renderEffortMenu();
              updateControls();
              closeAll();
            },
          })
        );
      }
    }

    if (
      !list.children.length
    ) {
      const empty =
        document.createElement(
          'div'
        );

      empty.className =
        'nai-empty';

      empty.textContent =
        MODELS.length
          ? 'No matching models.'
          : 'Loading the model list… if it stays empty, reload the AI chat.';

      list.appendChild(
        empty
      );
    }
  }

  // ---------------------------------------------------------------------------
  // Controls
  // ---------------------------------------------------------------------------

  function updateControls() {
    const label =
      document.querySelector(
        '#nai-model-trigger .nai-model-label'
      );

    if (label) {
      const id =
        selectedModelId();

      const model =
        modelById(id);

      const next =
        !id
          ? 'Auto'
          : model
            ? model.label
            : id;

      if (
        label.textContent !==
        next
      ) {
        label.textContent =
          next;
      }
    }

    const trigger =
      document.getElementById(
        'nai-effort-trigger'
      );

    if (trigger) {
      const options =
        availableEfforts();

      const active =
        currentEffort();

      const index =
        active
          ? options.indexOf(active)
          : -1;

      const filled =
        index < 0
          ? 0
          : Math.max(
              1,
              Math.round(
                (
                  (index + 1) /
                  options.length
                ) * 4
              )
            );

      trigger
        .querySelectorAll(
          '.nai-bars i'
        )
        .forEach(
          (
            bar,
            position
          ) => {
            bar.classList.toggle(
              'on',
              position <
                filled
            );
          }
        );

      trigger.title =
        active
          ? `Thinking effort: ${titleCase(active)}`
          : 'Thinking effort: controlled by Notion';
    }
  }

  // ---------------------------------------------------------------------------
  // Positioning
  // ---------------------------------------------------------------------------

  function triggerFor(menuId) {
    return document.getElementById(
      menuId ===
        EFFORT_MENU_ID
        ? 'nai-effort-trigger'
        : 'nai-model-trigger'
    );
  }

  function positionMenus() {
    for (
      const menuId of [
        MODEL_MENU_ID,
        EFFORT_MENU_ID,
      ]
    ) {
      const menu =
        document.getElementById(
          menuId
        );

      const trigger =
        triggerFor(menuId);

      if (
        !menu ||
        !trigger ||
        !menu.classList.contains(
          'open'
        )
      ) {
        continue;
      }

      const rect =
        trigger.getBoundingClientRect();

      const width =
        menu.offsetWidth ||
        280;

      const height =
        menu.offsetHeight ||
        320;

      const left =
        Math.max(
          12,
          Math.min(
            rect.right -
              width,
            window.innerWidth -
              width -
              12
          )
        );

      const openAbove =
        rect.top >=
        height + 12;

      const top =
        openAbove
          ? rect.top -
            height -
            8
          : Math.min(
              window.innerHeight -
                height -
                12,
              rect.bottom +
                8
            );

      menu.style.left =
        `${left}px`;

      menu.style.top =
        `${Math.max(
          12,
          top
        )}px`;
    }
  }

  function closeAll() {
    for (
      const menuId of [
        MODEL_MENU_ID,
        EFFORT_MENU_ID,
      ]
    ) {
      document
        .getElementById(
          menuId
        )
        ?.classList.remove(
          'open'
        );

      triggerFor(menuId)
        ?.setAttribute(
          'aria-expanded',
          'false'
        );
    }
  }

  function togglePopover(
    menuId
  ) {
    const menu =
      document.getElementById(
        menuId
      );

    const isOpen =
      menu?.classList.contains(
        'open'
      );

    closeAll();

    if (
      isOpen ||
      !menu
    ) {
      return;
    }

    if (
      menuId ===
      MODEL_MENU_ID
    ) {
      const search =
        document.getElementById(
          'nai-model-search'
        );

      if (search) {
        search.value = '';
      }

      renderModelList();

      if (!MODELS.length) {
        probeModelList();
      }

      menu.classList.add(
        'open'
      );

      triggerFor(menuId)
        ?.setAttribute(
          'aria-expanded',
          'true'
        );

      positionMenus();

      requestAnimationFrame(
        positionMenus
      );

      search?.focus({
        preventScroll: true,
      });

      return;
    }

    renderEffortMenu();

    menu.classList.add(
      'open'
    );

    triggerFor(menuId)
      ?.setAttribute(
        'aria-expanded',
        'true'
      );

    positionMenus();

    requestAnimationFrame(
      positionMenus
    );
  }

  // ---------------------------------------------------------------------------
  // Mount
  // ---------------------------------------------------------------------------

  function mount() {
    if (
      !document.body ||
      !document.head
    ) {
      return;
    }

    installStyle();

    const microphone =
      document.querySelector(
        MIC_SELECTOR
      ) ||
      document.querySelector(
        '[aria-label="Start voice recording"]'
      ) ||
      document
        .querySelector(
          'svg.microphone'
        )
        ?.closest(
          '[role="button"]'
        );

    const sendButton =
      document.querySelector(
        '[data-testid="agent-send-message-button"]'
      ) ||
      document.querySelector(
        '[aria-label="Submit AI message"]'
      );

    if (
      !microphone &&
      !sendButton
    ) {
      return;
    }

    const anchor =
      microphone
        ? (
            microphone.closest(
              '[data-popup-origin="true"]'
            ) ||
            microphone
          )
        : sendButton;

    const actionsRow =
      anchor?.parentElement;

    if (
      !anchor ||
      !actionsRow
    ) {
      return;
    }

    const modelHost =
      document.getElementById(
        MODEL_HOST_ID
      ) ||
      buildModelHost();

    const effortHost =
      document.getElementById(
        EFFORT_HOST_ID
      ) ||
      buildEffortHost();

    // [effort] [model] [microphone] [send]

    if (
      modelHost.parentElement !==
        actionsRow ||
      modelHost.nextSibling !==
        anchor
    ) {
      actionsRow.insertBefore(
        modelHost,
        anchor
      );
    }

    if (
      effortHost.parentElement !==
        actionsRow ||
      effortHost.nextSibling !==
        modelHost
    ) {
      actionsRow.insertBefore(
        effortHost,
        modelHost
      );

      console.log(
        '[ModelPicker] controls mounted in the composer toolbar'
      );
    }

    if (
      !document.getElementById(
        MODEL_MENU_ID
      )
    ) {
      buildModelMenu();
    }

    if (
      !document.getElementById(
        EFFORT_MENU_ID
      )
    ) {
      buildEffortMenu();
    }

    updateControls();

    if (!MODELS.length) {
      probeModelList();
    }
  }

  function scheduleMount() {
    if (mountScheduled) {
      return;
    }

    mountScheduled = true;

    requestAnimationFrame(
      () => {
        mountScheduled =
          false;

        mount();
      }
    );
  }

  // ---------------------------------------------------------------------------
  // Global events
  // ---------------------------------------------------------------------------

  document.addEventListener(
    'click',
    closeAll,
    true
  );

  document.addEventListener(
    'keydown',
    (event) => {
      if (
        event.key ===
        'Escape'
      ) {
        closeAll();
      }
    }
  );

  window.addEventListener(
    'resize',
    positionMenus,
    {
      passive: true,
    }
  );

  window.addEventListener(
    'scroll',
    positionMenus,
    {
      passive: true,
      capture: true,
    }
  );

  if (
    document.readyState ===
    'loading'
  ) {
    document.addEventListener(
      'DOMContentLoaded',
      mount,
      {
        once: true,
      }
    );
  } else {
    mount();
  }

  new MutationObserver(
    scheduleMount
  ).observe(
    document.documentElement,
    {
      childList: true,
      subtree: true,
    }
  );

  console.log(
    '[ModelPicker] loaded; waiting for getAvailableModels…'
  );
})();
